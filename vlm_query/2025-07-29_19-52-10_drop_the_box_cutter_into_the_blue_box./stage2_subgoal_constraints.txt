def stage2_subgoal_constraint1(end_effector, keypoints):
    """Position the end-effector 10cm above the center of the blue box (average of keypoints 3, 4, 5)."""
    box_center = (keypoints[3] + keypoints[4] + keypoints[5]) / 3
    offsetted_point = box_center + np.array([0, 0, 0.10])  # 10cm above
    cost = np.linalg.norm(end_effector - offsetted_point)
    return cost
