def stage2_subgoal_constraint1(end_effector, keypoints):
    """Position the end-effector 10cm above the center of the blue box (average of keypoints 4, 5, 6)."""
    center = (keypoints[4] + keypoints[5] + keypoints[6]) / 3
    offsetted_point = center + np.array([0, 0, 0.10])  # 10cm above
    cost = np.linalg.norm(end_effector - offsetted_point)
    return cost
