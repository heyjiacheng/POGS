def stage2_subgoal_constraint1(end_effector, keypoints):
    """The end-effector is 10cm above the center of the blue box (average of keypoints 3 and 4)."""
    center_point = (keypoints[3] + keypoints[4]) / 2
    offsetted_point = center_point + np.array([0, 0, 0.10])  # 10cm above
    cost = np.linalg.norm(end_effector - offsetted_point)
    return cost
