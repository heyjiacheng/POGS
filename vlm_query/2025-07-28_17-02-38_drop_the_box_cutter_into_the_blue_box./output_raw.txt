# 1. Grasp the box cutter.
# 2. Move the box cutter above the blue box.
# 3. Drop the box cutter into the blue box.

def stage1_subgoal_constraint1(end_effector, keypoints):
    """Align the end-effector with the handle of the box cutter (keypoint 0)."""
    target_point = keypoints[0]
    cost = np.linalg.norm(end_effector - target_point)
